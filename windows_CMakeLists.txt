# Helper function to add executables with common libraries
function(add_tru_executable TARGET_NAME SOURCE_FILE)
    add_executable(${TARGET_NAME}
        ${SOURCE_FILE}
        ${COMMON_SOURCES}
        ${COMMON_HEADERS}
    )
    
    target_link_libraries(${TARGET_NAME}
        PRIVATE
        OpenSSL::SSL
        OpenSSL::Crypto
        leveldb::leveldb
        fmt::fmt
        ${Protobuf_LIBRARIES}
        ${CURL_LIBRARIES}
        ${OPENCL_LIBRARIES}
        jsoncpp_lib
        ${CRC32C_LIBRARY}
        ${SODIUM_LIBRARIES}
        ${QRENCODE_LIBRARY}
        ${ETHASH_LIBRARY}
        ${KECCAK_LIBRARY}
        ${WINDOWS_LIBS}
    )
    
    if(NOT WIN32)
        target_link_libraries(${TARGET_NAME} PRIVATE pthread)
    else()
        # For Windows cross-compilation, use winpthreads
        find_library(PTHREAD_LIB NAMES pthread winpthread
            HINTS ${CMAKE_SYSROOT}/lib
            NO_DEFAULT_PATH
        )
        if(PTHREAD_LIB)
            target_link_libraries(${TARGET_NAME} PRIVATE ${PTHREAD_LIB})
        endif()
    endif()
    
    if(BUILD_WITH_QT)
        target_link_libraries(${TARGET_NAME} PRIVATE Qt5::Widgets)
    endif()
    
    if(WALLY_LIBRARY)
        target_link_libraries(${TARGET_NAME} PRIVATE ${WALLY_LIBRARY})
    endif()
    
    set_target_properties(${TARGET_NAME} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin
    )
    
    # Add static linking flags for specific executables
    if(${TARGET_NAME} STREQUAL "tru_miner_cpu" OR 
       ${TARGET_NAME} STREQUAL "tru_wallet" OR 
       ${TARGET_NAME} STREQUAL "blockexplorer")
        if(NOT WIN32)
            set_target_properties(${TARGET_NAME} PROPERTIES
                LINK_FLAGS "-static-libgcc -static-libstdc++"
            )
        endif()
    endif()
    
    # Debug options
    if(CMAKE_BUILD_TYPE STREQUAL "Debug")
        target_compile_options(${TARGET_NAME} PRIVATE -g -O0)
    endif()
    
    # Disable deprecation warnings
    target_compile_options(${TARGET_NAME} PRIVATE -Wno-deprecated-declarations)
endfunction()
